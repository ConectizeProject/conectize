# Copie para .env na mesma pasta e ajuste.

# ── Supabase (projeto só da Evolution): Settings → Database
# Preferência: connection string "Direct connection" para o container aplicar migrações Prisma.
# Inclua ?sslmode=require e deixe o schema igual ao exemplo (evolution_api).
#
# Opção A — conexão direta Supabase (só se db.xxxxx.supabase.co resolver na sua rede):
# DATABASE_PROVIDER=postgresql
# DATABASE_CONNECTION_URI="postgresql://postgres.xxxxx:SUA_SENHA@db.xxxxx.supabase.co:5432/postgres?schema=evolution_api&sslmode=require"
#
# Opção B — Session pooler Supabase (IPv4; obrigatório DATABASE_BOUNCER quando psql_bouncer):
DATABASE_PROVIDER=psql_bouncer
DATABASE_CONNECTION_URI="postgresql://postgres.xxxxx:SUA_SENHA@aws-0-REGIAO.pooler.supabase.com:5432/postgres?schema=evolution_api&sslmode=require"
DATABASE_BOUNCER_CONNECTION_URI="postgresql://postgres.xxxxx:SUA_SENHA@aws-0-REGIAO.pooler.supabase.com:5432/postgres?pgbouncer=true&schema=evolution_api&sslmode=require"
DATABASE_CONNECTION_CLIENT_NAME=conectize_evolution
#
# Senha com @ ou # → URL-encode (%40, %23). Use aspas duplas nas URIs.
# Com psql_bouncer, faltando DATABASE_BOUNCER_CONNECTION_URI → Prisma P1012 no Docker.
#
CACHE_REDIS_ENABLED=true
CACHE_REDIS_URI=redis://redis-evolution:6379/6
CACHE_REDIS_PREFIX_KEY=conectize_evolution
#
SERVER_NAME=evolution
SERVER_URL=https://SUBDOMINIO.ONDE.HOSPEDA.COM
SERVER_PORT=8080
#
